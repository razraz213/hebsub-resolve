# This set is 432 MB and resolves on Windows, Linux and macOS. The optional
# files below add GPU acceleration and one opt-in proofread pass; installing
# everything comes to 3.0 GB, and 2.5 GB of that buys either speed on an
# NVIDIA card or a feature measured at 0.12pp. Both are choices, not defaults.

faster-whisper==1.2.1

# Not optional, despite the size. `proofread.hebrew_lexicon` builds its
# 105k-word list from DictaBERT's TOKENIZER, which needs transformers but NOT
# torch -- verified in a venv with torch absent. That list is what the
# second-opinion rule judges against (D62: 6 words fixed, 0 broken), so
# dropping it does not fail, it silently degrades to flag-only.
transformers==5.15.1
